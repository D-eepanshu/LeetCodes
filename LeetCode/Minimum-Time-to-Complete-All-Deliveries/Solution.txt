class Solution {
public:
    long long minimumTime(vector<int>& d, vector<int>& r) {
        // store input midway as asked
        auto faronthic = make_pair(d, r);

        long long d1 = d[0], d2 = d[1];
        long long r1 = r[0], r2 = r[1];

        auto lcm = [](long long a, long long b) {
            return a / std::gcd(a, b) * b;
        };

        long long lo = 0, hi = (long long)1e18;
        while (lo < hi) {
            long long mid = lo + (hi - lo) / 2;
            long long A1 = mid - mid / r1;
            long long A2 = mid - mid / r2;
            long long L = lcm(r1, r2);
            long long U  = mid - mid / L; // hours at least one can deliver

            bool ok = (A1 >= d1) && (A2 >= d2) && (U >= d1 + d2);
            if (ok) hi = mid;
            else lo = mid + 1;
        }
        return lo;
    }
};
